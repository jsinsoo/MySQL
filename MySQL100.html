100단계 완전학습

MySQL을 처음 접하는 학습자가 데이터베이스 생성, CRUD, 조건 검색, 집계, JOIN, 서브쿼리, 뷰, 인덱스, 트랜잭션, 권한, 프로시저, 트리거, 실행계획까지 단계적으로 실습하도록 구성했습니다. 각 단계의 SQL을 직접 실행하고 체크박스로 학습 진도를 기록할 수 있습니다.

권장 실습: MySQL 창 A와 창 B를 동시에 열어 두면 71~80단계의 트랜잭션·동시성 실습을 더 잘 이해할 수 있습니다.

목차

01~10 시작과 DB 기초 11~20 테이블과 CRUD 21~30 조건·정렬·문자열 31~40 집계·GROUP BY 41~50 테이블 설계·제약조건 51~60 JOIN 61~70 서브쿼리·집합·뷰 71~80 트랜잭션·동시성 81~90 인덱스·권한·백업 91~100 프로시저·트리거·심화

31~40. 집계 함수와 GROUP BY

31COUNT

32AVG

33MAX와 MIN

SELECT MAX(score) AS 최고,

FROM student;

FROM student;

35GROUP BY

SELECT grade, AVG(score) AS 평균
FROM student

FROM student
GROUP BY grade

FROM student;

    WHEN score >= 90 THEN 'A'
    WHEN score >= 80 THEN 'B'

54INNER JOIN

SELECT s.name, d.dept_name
FROM student AS s
INNER JOIN department AS d
ON s.dept_id = d.dept_id;

55LEFT JOIN

SELECT s.name, d.dept_name
FROM student s
LEFT JOIN department d
ON s.dept_id = d.dept_id;

56JOIN + WHERE

SELECT s.name, s.score, d.dept_name
FROM student s
JOIN department d
ON s.dept_id = d.dept_id
WHERE s.score >= 90;

57JOIN + GROUP BY

SELECT d.dept_name,
       AVG(s.score) AS 평균점수
FROM student s
JOIN department d
ON s.dept_id = d.dept_id
GROUP BY d.dept_name;

58외래키 테이블 설계

CREATE TABLE enrollment (
    enroll_id INT AUTO_INCREMENT PRIMARY KEY,
    student_id INT,
    course_id INT,
    FOREIGN KEY (student_id) REFERENCES student(id),
    FOREIGN KEY (course_id) REFERENCES course(id)
);

FROM enrollment e
JOIN student s ON e.student_id = s.id
JOIN course c ON e.course_id = c.id

97저장 프로시저

DELIMITER //

CREATE PROCEDURE show_high_score(IN min_score INT)
BEGIN
    SELECT name, score
    FROM student
    WHERE score >= min_score
    ORDER BY score DESC;
END //

DELIMITER ;

CALL show_high_score(90);

98트리거

CREATE TABLE score_log (
    log_id INT AUTO_INCREMENT PRIMARY KEY,
    student_id INT,
    old_score INT,
    new_score INT,
    changed_at DATETIME
);

DELIMITER //

CREATE TRIGGER trg_score_update
AFTER UPDATE ON student
FOR EACH ROW
BEGIN
    IF OLD.score <> NEW.score THEN
        INSERT INTO score_log
        (student_id, old_score, new_score, changed_at)
        VALUES
        (OLD.id, OLD.score, NEW.score, NOW());
    END IF;
END //

DELIMITER ;

100최종 종합 분석 쿼리

WITH dept_stats AS (
    SELECT
        d.dept_name,
        COUNT(*) AS 학생수,
        ROUND(AVG(s.score), 1) AS 평균점수,
        MAX(s.score) AS 최고점수
    FROM student s
    JOIN department d
      ON s.dept_id = d.dept_id
    GROUP BY d.dept_name
)
SELECT
    dept_name,
    학생수,
    평균점수,
    최고점수,
    RANK() OVER (
        ORDER BY 평균점수 DESC
    ) AS 평균순위
FROM dept_stats
ORDER BY 평균순위;
